Starter Code  win32_window.c ▼
win32_window.c, build.bat: set SOURCES=win32_window.c, and user32.lib gdi32.lib on the cl line 
#include <windows.h>

// Global state: where the rectangle is
static int g_rect_x = 100;
static int g_rect_y = 100;
static HWND g_hwnd = NULL;

// Windows calls this function for every event (key press, mouse move, close, etc.)
LRESULT CALLBACK WindowProc(HWND hwnd, UINT msg, WPARAM wp, LPARAM lp) {
    switch (msg) {
        case WM_KEYDOWN:
            if (wp == VK_LEFT)  g_rect_x -= 5;
            if (wp == VK_RIGHT) g_rect_x += 5;
            if (wp == VK_UP)    g_rect_y -= 5;
            if (wp == VK_DOWN)  g_rect_y += 5;
            InvalidateRect(hwnd, NULL, TRUE); // Tell Windows to redraw
            break;
        case WM_PAINT: {
            PAINTSTRUCT ps;
            HDC hdc = BeginPaint(hwnd, &ps);
            FillRect(hdc, &ps.rcPaint, (HBRUSH)(COLOR_WINDOW+1)); // Clear
            RECT r;
            SetRect(&r, g_rect_x, g_rect_y, g_rect_x+50, g_rect_y+50); // left, top, right, bottom
            HBRUSH brush = CreateSolidBrush(RGB(0,200,100));
            FillRect(hdc, &r, brush);
            DeleteObject(brush);   // give the brush back, or every repaint leaks one
            EndPaint(hwnd, &ps);
        } break;
        case WM_DESTROY:
            PostQuitMessage(0);
            break;
        default:
            return DefWindowProc(hwnd, msg, wp, lp);
    }
    return 0;
}

int WINAPI WinMain(HINSTANCE hInst, HINSTANCE hPrev, LPSTR lpCmd, int nShow) {
    (void)hPrev; (void)lpCmd;   // unused: this stops the unused-parameter warning under /W4 /WX

    // Register the window class
    WNDCLASS wc = {0};
    wc.lpfnWndProc   = WindowProc;
    wc.hInstance     = hInst;
    wc.lpszClassName = "GameWindow";
    wc.hCursor       = LoadCursor(NULL, IDC_ARROW);
    RegisterClass(&wc);

    // Create the window
    g_hwnd = CreateWindowEx(0, "GameWindow", "My First Window",
                   WS_OVERLAPPEDWINDOW, CW_USEDEFAULT, CW_USEDEFAULT,
                   800, 600, NULL, NULL, hInst, NULL);
    ShowWindow(g_hwnd, nShow);

    // The message loop: this IS the game loop at the OS level.
    // PeekMessage never waits, so a frame runs whether or not a message came in.
    MSG msg = {0};
    int running = 1;
    while (running) {
        while (PeekMessage(&msg, NULL, 0, 0, PM_REMOVE)) {
            if (msg.message == WM_QUIT) running = 0;
            TranslateMessage(&msg);
            DispatchMessage(&msg);
        }
        // TODO: the frame timer goes here. Measure the frame, then sleep
        // only for what's left of 16.67 ms instead of this fixed guess.
        Sleep(16);
    }
    return (int)msg.wParam;
}

Stuck  Win32 specifics ▼
Linker errors about unresolved user32 or gdi32 symbols
Add the Windows libraries to build.bat's cl line: cl %FLAGS% %SOURCES% user32.lib gdi32.lib /Fe:game.exe. They hold the real code behind CreateWindowEx, GetMessage and friends.
The rectangle flickers when it moves
Expected with plain GDI. Double buffering is the fix, and SDL2 or OpenGL later removes it altogether.

Mixed Review  one from Phase 3.5, one from Phase 4.5 ▼
Two questions from earlier phases. Answer out loud first.
1. [Phase 3.5] Win32 has you write GetClientRect(hwnd, &rect);. Why the &?
Reveal answer
Because the function fills in your struct for you, and to change your struct it needs its address, exactly like player_move(&hero, …) and scanf("%d", &n). Whenever a Win32 function "returns" a struct through a parameter, it's writing through a pointer you handed it.
2. [Phase 4.5] Why did the runner build each frame in a char array before printing it, and where does that idea show up in this phase?
Reveal answer
So the whole frame appears at once instead of being drawn piece by piece, which is what flicker is. In a window it's called double buffering: draw everything to an off-screen bitmap, then copy it to the window in one go. Same idea, pixels instead of characters.
